cmake_minimum_required(VERSION 3.16)
project(quant_termux VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/external/quant-core/CMakeLists.txt)
  set(QUANT_CORE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/external/quant-core)
elseif(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/../quant-core/CMakeLists.txt)
  set(QUANT_CORE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../quant-core)
else()
  message(FATAL_ERROR "quant-core not found: run 'git submodule update --init'")
endif()
set(QUANT_BUILD_DAEMON OFF CACHE BOOL "" FORCE)
set(QUANT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(QUANT_UPNP OFF CACHE BOOL "" FORCE) # phones don't accept inbound connections
add_subdirectory(${QUANT_CORE_DIR} quant-core)

add_executable(qnt src/qnt.cpp qrcodegen/qrcodegen.cpp)
target_include_directories(qnt PRIVATE qrcodegen)
target_link_libraries(qnt PRIVATE quant)
if(WIN32)
  target_link_options(qnt PRIVATE -static)
endif()
install(TARGETS qnt DESTINATION bin)
